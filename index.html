/** * 數位科技概論｜幾A幾B雲端存檔・後端 (Code.gs) */ function doGet(e) { const action = e.parameter.action; if (action === 'loadGames') { const userId = e.parameter.userId; const result = apiLoadGames(userId); return createCorsResponse(result); } return HtmlService.createHtmlOutputFromFile('Index') .setTitle('幾A幾B大挑戰・雲端版') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function doPost(e) { try { let data; if (e.parameter && e.parameter.payload) { data = JSON.parse(e.parameter.payload); } else if (e.postData && e.postData.contents) { data = JSON.parse(e.postData.contents); } else { throw new Error("無效的請求資料"); } let result = { success: false, error: "未知動作" }; if (data.action === 'saveGame') { result = apiSaveGame(data.userId, data.gameState); } return createCorsResponse(result); } catch (err) { return createCorsResponse({ success: false, error: err.toString() }); } } function createCorsResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doOptions(e) { return ContentService.createTextOutput("") .setMimeType(ContentService.MimeType.TEXT); } // 初始化試算表與設定 function setup() { const ssName = "數位科技概論｜幾A幾B版本三存檔"; let sheetId = PropertiesService.getScriptProperties().getProperty('GUESS_BOOK_ID'); let ss; if (sheetId) { try { ss = SpreadsheetApp.openById(sheetId); } catch (e) { ss = null; } } if (!ss) { const files = DriveApp.getFilesByName(ssName); if (files.hasNext()) { ss = SpreadsheetApp.open(files.next()); } else { ss = SpreadsheetApp.create(ssName); let sheet = ss.getActiveSheet(); sheet.setName("遊戲存檔"); sheet.appendRow([ "使用者代號", "局ID", "局數", "狀態", "開始時間", "更新時間", "猜測次數", "猜測歷程", "完整存檔JSON" ]); } PropertiesService.getScriptProperties().setProperty('GUESS_BOOK_ID', ss.getId()); } let targetSheet = ss.getSheetByName("遊戲存檔"); if (!targetSheet) { let defaultSheet = ss.getSheetByName("工作表1"); if (defaultSheet) { defaultSheet.setName("遊戲存檔"); targetSheet = defaultSheet; } else { targetSheet = ss.insertSheet("遊戲存檔"); targetSheet.appendRow([ "使用者代號", "局ID", "局數", "狀態", "開始時間", "更新時間", "猜測次數", "猜測歷程", "完整存檔JSON" ]); } } ss.setSpreadsheetTimeZone("Asia/Taipei"); return ss.getUrl(); } function getSheet() { let sheetId = PropertiesService.getScriptProperties().getProperty('GUESS_BOOK_ID'); let ss; if (sheetId) { try { ss = SpreadsheetApp.openById(sheetId); } catch (e) { ss = null; } } if (!ss) { setup(); sheetId = PropertiesService.getScriptProperties().getProperty('GUESS_BOOK_ID'); ss = SpreadsheetApp.openById(sheetId); } let sheet = ss.getSheetByName("遊戲存檔"); if (!sheet) { setup(); sheet = ss.getSheetByName("遊戲存檔"); } return sheet; } // API: 讀取某個代號的所有遊戲局 function apiLoadGames(userId) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); userId = sanitizeUserId(userId); if (!userId) return { success: false, error: "代號無效" }; const sheet = getSheet(); const rows = sheet.getDataRange().getValues(); let games = []; for (let i = 1; i < rows.length; i++) { if (rows[i][0] === userId) { try { let data = JSON.parse(rows[i][8]); games.push(data); } catch (err) {} } } let uniqueGames = Array.from(new Set(games.map(g => g.gameId))) .map(id => games.find(g => g.gameId === id)); return { success: true, games: uniqueGames }; } catch (err) { return { success: false, error: err.toString() }; } finally { lock.releaseLock(); } } // API: 儲存或更新遊戲狀態 function apiSaveGame(userId, gameState) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); userId = sanitizeUserId(userId); if (!userId || !gameState || !gameState.gameId) { return { success: false, error: "資料不完整" }; } const sheet = getSheet(); const rows = sheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < rows.length; i++) { if (rows[i][0] === userId && rows[i][1] === gameState.gameId) { rowIndex = i + 1; break; } } gameState.updateTime = new Date().toISOString(); let jsonStr = JSON.stringify(gameState); if (rowIndex > -1) { sheet.getRange(rowIndex, 4).setValue(gameState.status); sheet.getRange(rowIndex, 6).setValue(gameState.updateTime); sheet.getRange(rowIndex, 7).setValue(gameState.guessesCount); sheet.getRange(rowIndex, 8).setValue(formatHistory(gameState.history)); sheet.getRange(rowIndex, 9).setValue(jsonStr); } else { sheet.appendRow([ userId, gameState.gameId, gameState.gameNum, gameState.status, gameState.startTime, gameState.updateTime, gameState.guessesCount, formatHistory(gameState.history), jsonStr ]); } return { success: true }; } catch (err) { return { success: false, error: err.toString() }; } finally { lock.releaseLock(); } } function sanitizeUserId(id) { if (!id) return ""; return id.toString().trim().toLowerCase().replace(/[^a-z0-9_\-]/g, ''); } function formatHistory(history) { if (!history || history.length === 0) return ""; return history.map((h, idx) => `${idx + 1}. ${h.guess} → ${h.result}`).join(" | "); }
